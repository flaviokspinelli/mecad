cmake_minimum_required(VERSION 3.24)
project(MecaCAD VERSION 0.2.18 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets OpenGLWidgets OpenGL Test)
find_package(OpenCASCADE REQUIRED)
add_library(mecacore src/model.cpp src/model.h)
target_include_directories(mecacore SYSTEM PUBLIC ${OpenCASCADE_INCLUDE_DIR})
target_link_directories(mecacore PUBLIC ${OpenCASCADE_LIBRARY_DIR})
target_link_libraries(mecacore PUBLIC Qt6::Widgets TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKBO TKBool TKMesh TKShHealing TKFillet TKDESTEP TKDESTL)
target_include_directories(mecacore PUBLIC src)
add_library(mecaui src/window.cpp src/window.h src/viewport.cpp src/viewport.h src/picking.cpp)
target_link_libraries(mecaui PUBLIC mecacore Qt6::OpenGLWidgets Qt6::OpenGL)
qt_add_executable(MecaCAD MACOSX_BUNDLE src/main.cpp)
target_link_libraries(MecaCAD PRIVATE mecaui)
if(APPLE)
    target_sources(MecaCAD PRIVATE assets/MecaCAD.icns)
    set_source_files_properties(assets/MecaCAD.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_target_properties(MecaCAD PROPERTIES MACOSX_BUNDLE_ICON_FILE MecaCAD.icns)
endif()
set_target_properties(MecaCAD PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER com.mecacad.desktop MACOSX_BUNDLE_BUNDLE_NAME MecaCAD MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION} MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
enable_testing()
qt_add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE mecacore Qt6::Test)
add_test(NAME core COMMAND core_tests)
qt_add_executable(ui_tests tests/ui_tests.cpp)
target_link_libraries(ui_tests PRIVATE mecaui Qt6::Test)
add_test(NAME ui COMMAND ui_tests)
install(TARGETS MecaCAD BUNDLE DESTINATION . RUNTIME DESTINATION bin)
