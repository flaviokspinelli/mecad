cmake_minimum_required(VERSION 3.24)
project(Mecad VERSION 0.2.25 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets OpenGLWidgets OpenGL Test)
find_package(OpenCASCADE REQUIRED)
add_library(mecacore src/model.cpp src/model.h src/chamfer.cpp src/parameters.cpp src/sketch_edit.cpp src/parametric_sketch.cpp src/recovery.cpp src/recovery.h src/dependencies.cpp src/dependencies.h)
target_include_directories(mecacore SYSTEM PUBLIC ${OpenCASCADE_INCLUDE_DIR})
target_link_directories(mecacore PUBLIC ${OpenCASCADE_LIBRARY_DIR})
target_link_libraries(mecacore PUBLIC Qt6::Widgets TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKBO TKBool TKMesh TKShHealing TKFillet TKDESTEP TKDESTL)
target_include_directories(mecacore PUBLIC src)
add_library(mecaui src/window.cpp src/window.h src/viewport.cpp src/viewport.h src/picking.cpp)
target_link_libraries(mecaui PUBLIC mecacore Qt6::OpenGLWidgets Qt6::OpenGL)
qt_add_executable(Mecad MACOSX_BUNDLE src/main.cpp)
target_link_libraries(Mecad PRIVATE mecaui)
if(APPLE)
    target_sources(Mecad PRIVATE assets/MecaCAD.icns)
    set_source_files_properties(assets/MecaCAD.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_target_properties(Mecad PROPERTIES
        MACOSX_BUNDLE_ICON_FILE MecaCAD.icns
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/assets/Mecad-Info.plist.in")
endif()
set_target_properties(Mecad PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER com.mecad.desktop MACOSX_BUNDLE_BUNDLE_NAME Mecad MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION} MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
enable_testing()
add_library(mecaexpressions src/expressions.cpp src/expressions.h)
target_link_libraries(mecaexpressions PUBLIC Qt6::Core)
target_include_directories(mecaexpressions PUBLIC src)
target_link_libraries(mecacore PUBLIC mecaexpressions)
qt_add_executable(expression_tests tests/expression_tests.cpp)
target_link_libraries(expression_tests PRIVATE mecaexpressions Qt6::Test)
add_test(NAME expressions COMMAND expression_tests)
set_tests_properties(expressions PROPERTIES LABELS "parameters;headless" TIMEOUT 60)
add_library(mecasketch src/sketch_system.cpp src/sketch_system.h src/snap_intersections.cpp src/snap_intersections.h)
target_link_libraries(mecasketch PUBLIC Qt6::Core)
target_include_directories(mecasketch PUBLIC src)
target_link_libraries(mecacore PUBLIC mecasketch)
qt_add_executable(sketch_system_tests tests/sketch_system_tests.cpp)
target_link_libraries(sketch_system_tests PRIVATE mecasketch Qt6::Test)
add_test(NAME sketch_system COMMAND sketch_system_tests)
set_tests_properties(sketch_system PROPERTIES LABELS "sketch;headless" TIMEOUT 60)
qt_add_executable(snap_tests tests/snap_tests.cpp)
target_link_libraries(snap_tests PRIVATE mecasketch Qt6::Test)
add_test(NAME snapping COMMAND snap_tests)
set_tests_properties(snapping PROPERTIES LABELS "sketch;headless" TIMEOUT 60)
qt_add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE mecacore Qt6::Test)
add_test(NAME core COMMAND core_tests)
qt_add_executable(corpus_tests tests/corpus_tests.cpp)
target_link_libraries(corpus_tests PRIVATE mecacore Qt6::Test)
add_test(NAME corpus COMMAND corpus_tests)
set_tests_properties(corpus PROPERTIES LABELS "corpus;headless" TIMEOUT 60)
set_tests_properties(core PROPERTIES LABELS "core;headless")
qt_add_executable(recovery_tests tests/recovery_tests.cpp)
target_link_libraries(recovery_tests PRIVATE mecacore Qt6::Test)
add_test(NAME recovery COMMAND recovery_tests)
set_tests_properties(recovery PROPERTIES LABELS "recovery;headless" TIMEOUT 60)
qt_add_executable(ui_tests tests/ui_tests.cpp)
target_link_libraries(ui_tests PRIVATE mecaui Qt6::Test)
add_test(NAME ui COMMAND ui_tests)
set_tests_properties(ui PROPERTIES LABELS "ui;interactive")
install(TARGETS Mecad BUNDLE DESTINATION . RUNTIME DESTINATION bin)
